    <!-- Supabase Integration Script -->
    <script>
        // Supabase Credentials Provided by You
        const SUPABASE_URL = 'https://yfioaxyzbrynurynvpdi.supabase.co';
        const SUPABASE_ANON_KEY = 'sb_publishable_CgXlu8jziGVz_OcPT_SiHg_Bx2re8dp';
        
        const _supabase = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

        // Switch Views (Shop <-> Admin)
        function switchView(view) {
            if(view === 'shop') {
                document.getElementById('shop-view').classList.remove('hidden');
                document.getElementById('admin-view').classList.add('hidden');
                fetchProducts();
            } else {
                document.getElementById('shop-view').classList.add('hidden');
                document.getElementById('admin-view').classList.remove('hidden');
            }
        }

        // Fetch Products from Supabase
        async function fetchProducts() {
            const grid = document.getElementById('products-grid');
            const spinner = document.getElementById('loading-spinner');
            spinner.style.display = 'block';
            grid.innerHTML = '';

            try {
                let { data: products, error } = await _supabase.from('products').select('*').order('id', { ascending: false });
                
                spinner.style.display = 'none';

                if (error) throw error;

                if (!products || products.length === 0) {
                    grid.innerHTML = '<p class="col-span-full text-center text-neutral-500 text-xs">No products found. Click ADMIN PANEL to add some!</p>';
                    return;
                }

                products.forEach(p => {
                    const imageUrl = p.image ? p.image : 'https://picsum.photos/seed/jewel/500/500';
                    const card = `
                        <div class="bg-[#1B1B1B] border border-neutral-800 rounded-lg overflow-hidden flex flex-col justify-between">
                            <img src="${imageUrl}" alt="${p.name}" class="w-full h-48 object-cover">
                            <div class="p-3 text-center flex flex-col justify-between flex-grow">
                                <div>
                                    <h4 class="text-xs text-white font-medium truncate">${p.name}</h4>
                                    <p class="text-[11px] text-[#D4AF37] font-semibold mt-1">${p.price}</p>
                                </div>
                                <button onclick="orderWhatsApp('${p.name}', '${p.price}')" class="mt-3 w-full bg-[#25D366] text-white text-[10px] font-bold py-2 rounded hover:opacity-90 transition">
                                    Order on WhatsApp
                                </button>
                            </div>
                        </div>
                    `;
                    grid.innerHTML += card;
                });
            } catch (err) {
                spinner.style.display = 'none';
                grid.innerHTML = `<p class="col-span-full text-center text-red-500 text-xs">Error loading products: ${err.message}</p>`;
            }
        }

        // Add Product Form Handler
        document.getElementById('add-product-form').addEventListener('submit', async function(e) {
            e.preventDefault();
            const name = document.getElementById('prod-name').value;
            const price = document.getElementById('prod-price').value;
            const image = document.getElementById('prod-image').value;
            const btn = document.getElementById('submit-btn');

            btn.disabled = true;
            btn.innerText = 'SAVING TO SUPABASE...';

            try {
                const { error } = await _supabase.from('products').insert([
                    { name: name, price: price, image: image ? image : null }
                ]);

                if (error) throw error;

                alert('Product added successfully!');
                document.getElementById('add-product-form').reset();
                switchView('shop');
            } catch (err) {
                alert('Error: ' + err.message);
            } finally {
                btn.disabled = false;
                btn.innerText = 'ADD PRODUCT TO WEBSITE';
            }
        });

        // WhatsApp Direct Order Function
        function orderWhatsApp(name, price) {
            const phone = '923001234567'; // Apna WhatsApp number yahan dalein
            const msg = `Assalam-o-Alaikum, Main yeh jewellery order karna chahta hoon:\n\n*Product:* ${name}\n*Price:* ${price}\n\nKindly details share karein.`;
            window.open(`https://wa.me/${phone}?text=${encodeURIComponent(msg)}`, '_blank');
        }

        // Load products on page load
        window.onload = fetchProducts;
